// Compare what was lost
console.group('❓ WHAT DIDN\'T MIGRATE');
const backup = GameEngine.state.bank.items_backup || {};
const migrated = new Set([
    ...Object.keys(GameEngine.state.bank.stackable),
    ...Object.values(GameEngine.state.bank.instanced).map(i => i.baseItemId)
]);

for (let itemId of Object.keys(backup)) {
    const baseId = itemId.replace(/_instance_.*$/, '').replace(/_common|_uncommon|_rare|_epic|_legendary$/, '');
    if (!migrated.has(itemId) && !migrated.has(baseId)) {
        console.log(`❌ Didn't migrate: ${itemId}`, backup[itemId]);
    }
}
console.groupEnd();
