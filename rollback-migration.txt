console.log('🔄 Rolling back dual-bank migration...');

// Restore old structure from backup
if (GameEngine.state.bank.items_backup) {
    GameEngine.state.bank.items = GameEngine.state.bank.items_backup;
    console.log(`✅ Restored ${Object.keys(GameEngine.state.bank.items).length} items from backup`);

    // Clear dual-bank structure

    SaveSystem.save();

} else {
    console.error('❌ No backup found!');
}
